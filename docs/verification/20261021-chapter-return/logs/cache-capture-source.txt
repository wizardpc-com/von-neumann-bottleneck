	print("Host: ",ui.instrument_host.get_global_rect())
	print("Window: ",ui.instrument_windows[&"cache"].get_global_rect())
	print("Applied cache lines: ",ui.current_cache_lines)
	print(("PASS: " if failures.is_empty() else "FAIL: ") + "cache input capture")
