extends SceneTree
var host
var captured: bool = false
func _init() -> void: call_deferred("start")
func start() -> void:
    if not OS.get_user_data_dir().contains("VonNeumannBottleneckChecks/"): quit(1); return
    root.size = Vector2i(1280,720)
    root.content_scale_size = root.size
    root.get_node("Localization").set_locale("zh_CN")
    root.get_node("GameMode").set_mode(&"test")
    root.get_node("TaskNavigation").pending = "chapter_4/fields"
    host = load("res://src/layout_chapter/layout_chapter.tscn").instantiate()
    root.add_child(host)
    get_tree_timer()
func get_tree_timer() -> void:
    await create_timer(600).timeout
    quit()
func _process(_delta: float) -> bool:
    if not is_instance_valid(host): return false
    if not captured and root.get_node("LayoutChapter").test_named.get("fields",{}).has("Native A") and is_instance_valid(host.score_window) and host.score_window.visible:
        captured = true
        capture.call_deferred()
    return false
func capture() -> void:
    await RenderingServer.frame_post_draw
    var report: Dictionary = {"design":host.design,"rows":host.score_window.rows,"game_named":root.get_node("LayoutChapter").game_named}
    var f = FileAccess.open("res://native-gameplay-result.json",FileAccess.WRITE)
    f.store_string(JSON.stringify(report,"  "))
    print("NATIVE: named design retained, rankings open, rows=",host.score_window.rows.size())
    print("NATIVE_IMAGE: ",host.score_window.get_texture().get_image().save_png("res://native-gameplay.png"))
